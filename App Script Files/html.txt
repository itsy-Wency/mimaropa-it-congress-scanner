<script>
  const DEPLOYED_WEB_APP_URL = "https://script.google.com/macros/s/AKfycbx6VoSK_DH2cglmh95detPoEhpy1TD7xt6T1u289S8XXRsOudVpYhvbkOJDRITo9rtr/exec";
  let html5QrCode;
  let isProcessing = false;
  let lastScannedCode = "";
  let lastScanTime = 0;

  window.onload = function() {
    initScanner();
    loadAttendeeSuggestions();
  };

  function playSound(type) {
    try {
      const ctx = new (window.AudioContext || window.webkitAudioContext)();
      const osc = ctx.createOscillator();
      const gain = ctx.createGain();
      osc.connect(gain);
      gain.connect(ctx.destination);
      if (type === 'success') {
        osc.frequency.setValueAtTime(880, ctx.currentTime);
        gain.gain.setValueAtTime(0.2, ctx.currentTime);
        osc.start(); osc.stop(ctx.currentTime + 0.12);
      } else {
        osc.frequency.setValueAtTime(220, ctx.currentTime);
        gain.gain.setValueAtTime(0.3, ctx.currentTime);
        osc.start(); osc.stop(ctx.currentTime + 0.25);
      }
    } catch (e) {}
  }

  function loadAttendeeSuggestions() {
    if (typeof google !== 'undefined' && google.script && google.script.run) {
      google.script.run.withSuccessHandler(populateDatalist).getRegisteredAttendees();
    }
  }

  function populateDatalist(list) {
    const datalist = document.getElementById('attendeesDatalist');
    datalist.innerHTML = '';
    if (!list) return;
    list.forEach(item => {
      const opt = document.createElement('option');
      opt.value = item.id;
      opt.label = `${item.id} (${item.name})`;
      datalist.appendChild(opt);
    });
  }

  function initScanner() {
    html5QrCode = new Html5Qrcode("reader");
    html5QrCode.start(
      { facingMode: "environment" },
      { fps: 15, qrbox: { width: 220, height: 220 } },
      (decodedText) => {
        const now = Date.now();
        if (decodedText === lastScannedCode && (now - lastScanTime) < 2500) return;
        
        lastScannedCode = decodedText;
        lastScanTime = now;
        
        document.getElementById('searchInput').value = decodedText.trim().toUpperCase();
        processCheckIn(decodedText.trim().toUpperCase());
      },
      () => {}
    ).catch(err => {
      updateStatus("⚠️ Camera initialization failed.", "error");
    });
  }

  function handleKeyPress(e) { if (e.key === 'Enter') submitSearch(); }

  function submitSearch() {
    const val = document.getElementById('searchInput').value.trim().toUpperCase();
    if (!val) return updateStatus("Please enter an Attendee ID.", "error");
    processCheckIn(val);
  }

  function processCheckIn(attendeeId) {
    if (isProcessing) return;
    isProcessing = true;

    const session = document.querySelector('input[name="session"]:checked').value;
    updateStatus(`⚡ Verifying "${attendeeId}"...`, "idle");

    if (typeof google !== 'undefined' && google.script && google.script.run) {
      google.script.run
        .withSuccessHandler(handleResponse)
        .withFailureHandler(handleError)
        .processScan(attendeeId, session);
    } else {
      fetch(DEPLOYED_WEB_APP_URL, {
        method: 'POST',
        headers: { 'Content-Type': 'text/plain;charset=utf-8' },
        body: JSON.stringify({ attendeeId: attendeeId, session: session })
      })
      .then(res => res.json())
      .then(handleResponse)
      .catch(handleError);
    }
  }

  function handleResponse(res) {
    if (res.status === 'SUCCESS') {
      playSound('success');
      updateStatus(res.message, "success");
      document.getElementById('searchInput').value = "";
    } else if (res.status === 'ALREADY_PROCESSED') {
      playSound('error');
      updateStatus(res.message, "already");
    } else {
      playSound('error');
      updateStatus(res.message, "error");
    }
    setTimeout(() => { isProcessing = false; }, 400);
  }

  function handleError(err) {
    playSound('error');
    updateStatus("❌ Connection busy. Tap Submit or Scan again.", "error");
    setTimeout(() => { isProcessing = false; }, 400);
  }

  function updateStatus(msg, type) {
    const box = document.getElementById('statusBox');
    box.className = "status-box " + type;
    box.innerHTML = msg;
  }
</script>
